# 4.3 QUERY FUNCTION
# Example: filter penguins heavier than X grams


filter_penguins_by_weight <- function(data, weight_threshold) {
  
  result <- data |>
    filter(body_mass_g > weight_threshold) |>
    select(species, island, body_mass_g)
  
  return(result)
}

# Summary report for one species


generate_report <- function(data, species_name) {
  
  # Filter for selected species
  species_data <- data |>
    filter(species == species_name)
  
  # Summary statistics
  avg_mass <- mean(species_data$body_mass_g, na.rm = TRUE)
  avg_bill_length <- mean(species_data$bill_length_mm, na.rm = TRUE)
  min_flipper <- min(species_data$flipper_length_mm, na.rm = TRUE)
  max_flipper <- max(species_data$flipper_length_mm, na.rm = TRUE)
  record_count <- nrow(species_data)
  
  # Print formatted report
  cat("\n--------------------------------------\n")
  cat(" Penguin Report for:", species_name, "\n")
  cat("--------------------------------------\n")
  cat(" Records found:", record_count, "\n")
  cat(" Average body mass:", round(avg_mass, 1), "g\n")
  cat(" Average bill length:", round(avg_bill_length, 1), "mm\n")
  cat(" Minimum flipper length:", min_flipper, "mm\n")
  cat(" Maximum flipper length:", max_flipper, "mm\n")
  cat("--------------------------------------\n\n")
}

# Call reporting function for two species
generate_report(my_data, "Adelie")
